const objects=new Map(),failures=new Map(),counts={},certificates={},assertions={};
const progress=e=>console.log('# SOURCE_PROGRESS '+JSON.stringify({...e,at:new Date().toISOString(),rssBytes:process.memoryUsage().rss}));
const context={onProgress:progress};
function once(name,build){
  if(failures.has(name))throw failures.get(name);
  if(objects.has(name))return objects.get(name);
  counts[name]=(counts[name]??0)+1;progress({stage:name+'-start'});
  try{const result=build();objects.set(name,result);progress({stage:name+'-complete',nodeCount:result.G.nodes.length});return result;}
  catch(error){failures.set(name,error);throw error;}
}
const family=()=>once('commonFamily',()=>prepareActualCommonAdmissibleFamily({throughOrder:2,derivativeOrder:3,maxNodes:5000000,maxProfileJets:60000,maxJetRequests:600000},context));
const band=()=>once('ownBand',()=>family().constructBand({maxNodes:5000000},context));
const physical=()=>once('actualPhysicalCurl',()=>attachActualCommonBandPhysicalCurl(family(),band(),context));
const target=()=>once('actualTargetN1',()=>family().constructTargetWeightedNorms({derivativeOrder:1,maxNodes:5000000},context));
const globalWave=()=>once('globalWave',()=>attachActualCommonGlobalWaveBounds(family(),{targetWeightedNorms:target(),physicalCurl:physical()},{derivativeOrder:0,flatOrder:4},context));
const covariance=()=>once('globalCovariance',()=>attachActualCommonGlobalCovariance(globalWave(),context));
const theorem=()=>once('sameBackgroundFlatness',()=>attachActualBackgroundResidualFlatness(family(),{truncationOrder:2,derivativeOrder:0,decayOrder:1},context));
const actualEB=()=>once('sameActualEB',()=>attachActualCommonBandBackgroundResidual(theorem(),physical(),context));
async function persist(complete=false){
  if(process.env.M2_COMMON_PHYSICAL_EVIDENCE)await writeFile(process.env.M2_COMMON_PHYSICAL_EVIDENCE,JSON.stringify({
    schema:'MathScope.ActualCommonPhysicalPackageSourceExecution/1',complete,certificates,assertions,sourceConstructions:counts,
    executedSourcePrefix:2,executedBackgroundSlowOrder:3,executedTargetWeightedOrder:1,executedGlobalWaveOrder:0,executedFlatOrder:4,
    allInfiniteGraphsLiterallyExecuted:false,wholeM2Complete:false,M3Started:false,currentRSSBytes:process.memoryUsage().rss,
  },null,2)+'\n');
}

  assert.equal(f.source.pressureStressSeminorms.request.derivativeOrder,2);

  assert.equal(b.background.pressureAxisDatumAndCentrifugalPrimitiveRetained,true);
  certificates.family=await actualCommonAdmissibleFamilyCertificate(f);certificates.band=await actualCommonAdmissibleBandCertificate(b);await persist();
});

test('each sign has actual weighted third jets, complete Cartesian curl and guarded physical compositions',async()=>{
  const p=physical(),G=p.G;assert.equal(p.family,family());assert.equal(p.band,band());assert.equal(p.rows.length,2);
  assert.equal(p.checks.every(c=>c.pass),true);assert.equal(p.geometry.checks.every(c=>c.pass),true);
  for(const row of p.rows){
    assert.equal(row.jets.checks.length,19);assert.equal(row.jets.checks.every(c=>c.pass),true);
    assert.equal(row.composition.length,40);assert.equal(row.primaryComposition.length,3);
    assert.equal(row.originalWeight,p.band.target.weights[p.rows.indexOf(row)]);
    for(const binding of [...row.jets.potentialBindings,row.jets.pressureBinding,row.jets.radialFastBinding]){
      assert.equal(binding.originalBodyBound,true);assert.equal(binding.partials.every(x=>x.pass),true);
    }
    for(const c of [...row.composition,...row.primaryComposition]){
      assert.equal(c.exactBodyRetained,true);assert.equal(c.auxiliaryCoordinatesEliminated,true);assert.equal(c.zeroGuardBeforeSingularEvaluation,true);
    }
    assert.equal(row.temporal.originalFullCurlBeforeTimeCutoffDerivative,true);assert.equal(row.temporal.divisionByPsiUsed,false);
    assert.equal(row.temporal.checks.every(c=>c.pass),true);assert.equal(row.phaseChecks.every(c=>c.pass),true);
    assert.equal(row.physicalCurlRemainder.length,3);
  }
  assert.equal(p.variations.every(x=>x.allFortyComponents&&x.originalMatrixAndDatumRetained&&!x.finiteSumReplacesLimit),true);
  for(let j=0;j<3;j++){
    assert.equal(p.roots.actualCurlRemainder[j],G.sub(p.roots.combinedPhysicalJets.velocity[j],p.roots.actualPrimaryWave[j]));
    assert.equal(p.roots.retainedPhysicalResidualAfterTemporalForce[j],G.sub(p.roots.physicalResidual[j],p.roots.actualTemporalCutoffForce[j]));
  }
  assert.equal(p.residual.checks.every(x=>x.pass),true);assert.equal(p.scope.totalUncorrectedNSResidualDeclaredFlat,false);
  certificates.physical=await actualCommonBandPhysicalCurlCertificate(p);await persist();
});

test('the original all-label primary covariance and signed inverse retain the full physical residual for N6',async()=>{
  const c=covariance(),G=c.G;assert.equal(c.G,physical().G);assert.equal(c.family,family());
  assert.equal(c.checks.every(x=>x.pass),true);assert.equal(c.assembly.rows.length,81);
  assert.equal(c.local.squareRootRule.every(x=>x.nodeIsThePositiveSquareRoot),true);

  assert.deepEqual(c.n6Interface.fullPhysicalResidual,physical().roots.physicalResidual);
  assert.deepEqual(c.n6Interface.actualCurlRemainder,physical().roots.actualCurlRemainder);

test('all final private packages reject copies and stay stable under append without prematurely closing M2',async()=>{
  const p=physical(),g=globalWave(),c=covariance();actualEB();
  for(const [object,check]of [[p,assertActualCommonBandPhysicalCurl],[g,assertActualCommonGlobalWaveBounds],[c,assertActualCommonGlobalCovariance]]){
    assert.equal(check(object),true);assert.throws(()=>check({...object}),{code:'INVALID_SOURCE_CONSTRUCTION'});
  }
  p.G.fresh('after_common_physical_package');g.G.fresh('after_common_global_norms');
  assert.equal((await actualCommonBandPhysicalCurlCertificate(p)).graph.sha256,certificates.physical.graph.sha256);
  assert.equal((await actualCommonGlobalWaveBoundCertificate(g)).graph.sha256,certificates.globalWave.graph.sha256);
  assert.equal((await actualCommonGlobalCovarianceCertificate(c)).graph.sha256,certificates.covariance.graph.sha256);
  for(const object of [p,g,c,theorem(),actualEB()])assert.equal(object.scope.wholeM2Complete,false);
  for(const count of Object.values(counts))assert.equal(count,1);
  await persist(true);
});
