/** For a coordinate, the source dependency recurrence is exactly membership
 * in the free-coordinate set. In particular, an integral removes its own
 * bound coordinate from its body but includes both endpoint dependencies.
 * Cache one set per node instead of millions of (node, fresh-dummy) pairs.
 * Noncoordinate structural-subexpression queries retain the old rule.
 */
export class ActualInnerCovarianceExpressions extends ActualAnnulusCovarianceExpressions{
  dependsOn(id,variable){
    if(id===variable)return true;
    if(this.nodes[variable]?.op==='coordinate')return this.freeCoordinates(id).has(variable);
    return super.dependsOn(id,variable);
  }
}
function requestOf(input={}){
  if(!input||typeof input!=='object'||Array.isArray(input)||Object.keys(input).some(k=>!['sourceProfile','jetOrder','vanishingOrder'].includes(k)))
    fail('INVALID_INPUT','Use only the pinned source profile and finite scalar derivative orders. Fields, endpoints, H entries and positive lower premises cannot be supplied.');
  const sourceProfile=input.sourceProfile??SOURCE_PROFILE_ID,jetOrder=input.jetOrder??3,vanishingOrder=input.vanishingOrder??4;assertSourceProfile(sourceProfile);
  if(!Number.isSafeInteger(jetOrder)||jetOrder<0||!Number.isSafeInteger(vanishingOrder)||vanishingOrder<0)
    fail('INVALID_INPUT','Derivative and flat orders must be nonnegative finite integers.');
  if(jetOrder>12||vanishingOrder>256)fail('RESOURCE_LIMIT','This actual-source packet materializes scalar jets through12 and flat orders through256. No omitted derivative is zero.');
  return {sourceProfile,jetOrder,vanishingOrder};
}

export function prepareActualInnerCovarianceFlatExtension(input={},context={}){
  const request=requestOf(input),selected=prepareActualAdmissibleAnnulusCovariance({sourceProfile:request.sourceProfile},context);
  return attachActualInnerCovarianceFlatExtension(selected,request,context);
}
