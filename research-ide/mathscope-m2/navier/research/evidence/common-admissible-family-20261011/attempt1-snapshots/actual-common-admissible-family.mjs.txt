/** Generic finite recurrence, NOT an actual source certificate. Each
 * bound must be positive. Source callers authenticate every operand first.
 */
export function commonRecursiveSelector(G,{historicalRows,clauses,ellMinimum}){
  if(!G||!Array.isArray(historicalRows)||!historicalRows.length||!Array.isArray(clauses)||clauses.length!==historicalRows.length||
    !Number.isSafeInteger(ellMinimum)||!G.nodes[ellMinimum])fail('INVALID_INPUT','Use finite explicit coefficient bounds and the original band threshold.');
  const q=(n,d=1)=>G.q(n,d),h=G.parameter('h'),log2=G.log(q(2)),rows=[];let previous=G.zero;
  for(let n=1;n<=historicalRows.length;n++){
    const old=historicalRows[n-1];
    if(old.order!==n||!Number.isSafeInteger(old.logScale)||!G.nodes[old.logScale]||!Array.isArray(clauses[n-1])||!clauses[n-1].length)
      fail('INVALID_INPUT','Do not omit a preceding scale or its actual norm clauses.');
    const constraints=clauses[n-1].map(c=>{
      if(!Number.isSafeInteger(c.derivativeOrder)||c.derivativeOrder<0||c.derivativeOrder>n||!Number.isSafeInteger(c.bound)||!G.nodes[c.bound])
        fail('INVALID_INPUT','Every finite diagonal clause needs its own bound through m<=n.');
      return {...c,requiredLog:G.div(G.add(G.mul(q(n),log2),G.log(c.bound)),G.mul(q(n),h))};
    }),growth=G.add(previous,G.log(q(n===2?16:2))),smallBand=n===1?G.mul(G.add(ellMinimum,q(10)),log2):G.zero,
      logScale=G.maximum(old.logScale,growth,smallBand,...constraints.map(c=>c.requiredLog)),scale=G.exp(logScale);
    rows.push({order:n,historicalScale:old.scale,historicalLogScale:old.logScale,previousLogScale:previous,
      sourceCoefficientRoots:old.sourceCoefficientRoots,growth,smallBand,constraints,logScale,scale});previous=logScale;
  }
  return {rows,sourceAuthenticated:false,recursiveDefinition:{
    first:'log b1=max(log c1, log2, (ellMinimum+10)log2, (log2+log B_1m)/h for all source clauses m<=1)',
    second:'log b2=max(log c2, log b1+log16, (2log2+log B_2m)/(2h) for all source clauses m<=2)',
    later:'log b_n=max(log c_n, log b_(n-1)+log2, (nlog2+log B_nm)/(nh) for every source clause m<=n)',
    requiredInequality:'B_nm*b_n^(-n*h)<=2^-n; on every nonzero cutoff derivative q<=1/b_n, so B_nm*q^(2*n*h)<=2^-n*q^(n*h).',
    localFiniteness:'b_n>=2^n. For every compact subset of q>0 all sufficiently large cutoff terms and their derivatives are exactly zero.',
    prefixIndependence:'Each B_nm is the same source function/FTC expression for fixed n,m. A larger finite query adds derivatives and orders; it does not change those mathematical constants. All predecessors are used in the displayed recursion.',
    n1WeightedQualification:'The original first weighted stress retains its explicit endpoint pole. A diagonal bound on its constant does not remove that pole.',
    noSimplePointwiseMaximumOfIndependentSequences:true,mathematicalOrderCap:null}};
}

function requestOf(input={}){
  if(!input||typeof input!=='object'||Array.isArray(input)||Object.keys(input).some(k=>!['sourceProfile','throughOrder','derivativeOrder','maxNodes','maxProfileJets','maxJetRequests'].includes(k)))
    fail('INVALID_INPUT','Only the source profile and finite query/representation budgets may be chosen.');
  const r={sourceProfile:input.sourceProfile??SOURCE_PROFILE_ID,throughOrder:input.throughOrder??2,derivativeOrder:input.derivativeOrder??2,
    maxNodes:input.maxNodes??4000000,maxProfileJets:input.maxProfileJets??60000,maxJetRequests:input.maxJetRequests??600000};
  assertSourceProfile(r.sourceProfile);
  for(const [k,v]of Object.entries(r))if(k!=='sourceProfile'&&(!Number.isSafeInteger(v)||v<(k==='derivativeOrder'?0:1)))
    fail('INVALID_INPUT','Use nonnegative finite derivative order and positive finite query budgets.');
  if(r.throughOrder<2||r.derivativeOrder<2)fail('RESOURCE_LIMIT','The common covariance constructor requires the genuine n1/n2 and slow C2 prefixes.');
  return r;
}

export function assertActualCommonAdmissibleFamily(p){
  const s=seals.get(p);
  if(!s||p.G!==s.G||p.prepared!==s.prepared||p.source.backgroundNorms!==s.backgroundNorms||p.source.velocitySeminorms!==s.velocitySeminorms||
    p.source.pressureStressSeminorms!==s.pressureStressSeminorms||p.constructBand!==s.constructBand||p.constructTargetWeightedNorms!==s.constructTargetWeightedNorms||
    data(p)!==s.data||canonicalStringify(p.program)!==s.program||canonicalStringify(p.G.nodes.slice(0,s.count))!==s.prefix||!p.G.functionDefinitionsUnchanged(s.definitions))
    fail('INVALID_SOURCE_CONSTRUCTION','Use the unchanged private common family, source graph, diagonal sequence and constructors.');
  assertActualAllBandBackgroundNorms(s.backgroundNorms);assertActualVelocityPhysicalSeminorms(s.velocitySeminorms);assertActualPressureStressPhysicalSeminorms(s.pressureStressSeminorms);return true;
}
