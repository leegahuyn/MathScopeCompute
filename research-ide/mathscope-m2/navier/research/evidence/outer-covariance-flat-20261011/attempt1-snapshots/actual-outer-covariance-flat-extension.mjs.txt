function finiteOrder(n,name,maximum=128){
  if(!Number.isSafeInteger(n)||n<0)fail('INVALID_INPUT',name+' must be a nonnegative safe integer.');
  if(n>maximum)fail('RESOURCE_LIMIT','The requested finite '+name+' exceeds this materialization budget. No omitted derivative is zero.');
  return n;
}
function requestOf(input={}){

  return {sourceProfile,jetOrder:finiteOrder(input.jetOrder??3,'jetOrder',12),vanishingOrder:finiteOrder(input.vanishingOrder??4,'vanishingOrder',256)};
}

/** Exact scalar statement. No source authentication is conferred here.
 * f(delta)=exp(-2/delta^2)*delta^(-3/2), delta>0; f=0 for delta<=0.
 * f^(j)=f*delta^(-3j)*P_j(delta^2).
 */
export function outerFlatSquareRootJets({order=3,vanishingOrder=4}={}){
  finiteOrder(order,'order');finiteOrder(vanishingOrder,'vanishingOrder',4096);
  let polynomial=[Q(1)];const rows=[];
  for(let j=0;j<=order;j++){
    let B=Q(0),quarterPower=Q(1);
    for(const c of polynomial){B=qadd(B,qmul(abs(c),quarterPower));quarterPower=qmul(quarterPower,Q(1,4));}
    const tradingInteger=Math.ceil((3+6*j+2*vanishingOrder)/4),coefficient=qmul(B,Q(factorial(tradingInteger)));
    rows.push({order:j,polynomial:polynomial.map(qtext),singularPower:qtext(Q(3+6*j,2)),
      polynomialAbsoluteUpperOnHalf:qtext(B),tradingInteger,vanishingOrder,coefficient:qtext(coefficient),
      bound:'coefficient * delta^vanishingOrder * exp(-1/delta^2)',
      exponentCheck:2*tradingInteger*2>=3+6*j+2*vanishingOrder});
    if(j<order){
      const next=Array.from({length:polynomial.length+1},()=>Q(0)),a=Q(3+6*j,2);
      polynomial.forEach((c,k)=>{next[k]=qadd(next[k],qmul(Q(4),c));next[k+1]=qadd(next[k+1],qmul(qsub(Q(2*k),a),c));});
      polynomial=next;
    }
  }
  return {schema:'MathScope.OuterFlatSquareRootJets/1',order,vanishingOrder,domain:'0<delta<=1/2; zero extension for delta<=0',rows,
    recurrence:'P_(j+1)(x)=[4-(3/2+3j)*x]*P_j(x)+2*x^2*P_j_prime(x)',
    tradingProof:'exp(1/delta^2)>=delta^(-2M)/M!, so exp(-1/delta^2)<=M!*delta^(2M). Since 2M>=3/2+3j+N and delta<=1, the stated bound follows.',
    allFiniteZeroJets:'For each fixed j choose N=1. The right derivative f^(j) tends to0, so extension by0 has that derivative at0. Induct on j; the same recurrence applies at every finite order.',
    arithmetic:'EXACT_BIGINT_RATIONAL',actualSourceAuthenticated:false,finiteOrderSamplesUsedAsAllOrderProof:false,
    pass:rows.every(r=>r.exponentCheck)};
}

/** A conditional, finite exact-rational sqrt jet recurrence. B[j] bounds
 * every ordinary multiindex derivative of W of total degree j. The lower
 * and upper values are premises, never actual-source certificates.
 */
export function positiveCoefficientSquareRootJetBounds({lower,upper,derivativeBounds}={}){
  const lo=readRational(lower),hi=readRational(upper);
  if(qcompare(lo,Q(0))<=0||qcompare(hi,lo)<0||!Array.isArray(derivativeBounds)||derivativeBounds.length<1)
    fail('INVALID_INPUT','Use a positive coefficient interval and every requested derivative bound.');
  finiteOrder(derivativeBounds.length-1,'derivative order');
  const B=derivativeBounds.map(x=>readRational(x));
  if(B.some(x=>qcompare(x,Q(0))<0)||qcompare(B[0],hi)<0)fail('INVALID_INPUT','Every derivative norm is nonnegative and B[0] must bound the supplied upper value.');
  const rootLower=qcompare(lo,Q(1))<0?lo:Q(1),bounds=[qadd(Q(1),hi)];
  for(let j=1;j<B.length;j++){
    let sum=B[j];for(let k=1;k<j;k++)sum=qadd(sum,qmul(Q(choose(j,k)),qmul(bounds[k],bounds[j-k])));
    bounds.push(qdiv(sum,qmul(Q(2),rootLower)));
  }
  return {schema:'MathScope.PositiveCoefficientSquareRootJetBounds/1',lower:qtext(lo),upper:qtext(hi),
    squareRootLower:qtext(rootLower),inputOrdinaryDerivativeBounds:B.map(qtext),ordinaryDerivativeBounds:bounds.map(qtext),
    recurrence:'2*sigma*D^alpha sigma=D^alpha W-sum_(0<beta<alpha) binom(alpha,beta) D^beta sigma D^(alpha-beta) sigma.',
    grouping:'sum_(|beta|=k,beta<=alpha) binom(alpha,beta)=binom(|alpha|,k). Neither endpoint beta=0 nor beta=alpha is included in the sum.',
    actualSourceAuthenticated:false,conditionalNormPremises:true,pass:true};
}

/** Exact exponent induction, independent of unspecified numeric constants.
 * It verifies the powers used AFTER actual smoothness, compact primitive
 * bounds, nonzero denominators and the logarithmic-norm lemma are bound.
 */
export function outerCovarianceJetDegreeAudit({order=8}={}){
  finiteOrder(order,'order');const rows=[];
  for(let j=0;j<=order;j++){
    const forcing=Array.from({length:j},(_,r)=>{const k=r+1;return {k,wExponent:1+k+2*(j-k),wUpper:2*j,
      coefficientExponent:1+2*k+3*(j-k)+1,coefficientUpper:3*j+1};});
    const squareRootProducts=Array.from({length:Math.max(0,j-1)},(_,r)=>{const k=r+1;return {k,exponent:4*k+4*(j-k),upper:4*j};});
    rows.push({order:j,matrixJetExponent:j,basisJetExponent:j,normalizedPulseJetExponent:2*j,covarianceJetExponent:2*j,
      inverseCoefficientJetExponent:3*j+1,squareRootCoefficientJetExponent:j?4*j:1,zeroExtendedProductExponent:4*j+1,
      forcing,squareRootProducts,pass:forcing.every(r=>r.wExponent<=r.wUpper&&r.coefficientExponent<=r.coefficientUpper)&&
        squareRootProducts.every(r=>r.exponent<=r.upper)&&(j===0||3*j+1<=4*j)});
  }
  return {schema:'MathScope.OuterCovarianceJetDegreeAudit/1',order,rows,pass:rows.every(r=>r.pass),
    normConstantsComputedHere:false,actualSourceAuthenticated:false,
    universalInduction:'The displayed inequalities hold for every j>=1 and 1<=k<=j. Finite rows are regression checks of these same affine identities, not extrapolation of observed derivative values.'};
}

export function logarithmicPulseNormAudit({operatorError='1',dampingError='1',lambda='1',scale='2',errorEntries=['1/2','1/2','1/2','1/2'],dampingDifference='-1/2',referenceSecondSign=-1}={}){
  const E=readRational(operatorError),Ed=readRational(dampingError),l=readRational(lambda),S=readRational(scale),dd=readRational(dampingDifference);
  if(qcompare(E,Q(0))<0||qcompare(Ed,Q(0))<0||qcompare(l,Q(0))<0||qcompare(S,Q(0))<=0||!Array.isArray(errorEntries)||errorEntries.length!==4||![-1,1].includes(referenceSecondSign))
    fail('INVALID_INPUT','Use the complete real 2x2 logarithmic-norm audit data.');
  const e=errorEntries.map(x=>readRational(x)),allowance=qdiv(E,S),dallowance=qdiv(Ed,S),premises=e.every(x=>qcompare(abs(x),allowance)<=0)&&qcompare(abs(dd),dallowance)<=0;
  const rows=[qadd(qsub(e[0],dd),abs(e[1])),qadd(qadd(qmul(Q(2*referenceSecondSign),l),qsub(e[3],dd)),abs(e[2]))],
    upper=qdiv(qadd(qmul(Q(2),E),Ed),S),mu=rows.reduce((a,b)=>qcompare(a,b)>0?a:b);
  return {schema:'MathScope.LogarithmicPulseNormAudit/1',rowLogarithmicNorms:rows.map(qtext),muInfinity:qtext(mu),upper:qtext(upper),
    sourceDampingReferenceSign:referenceSecondSign,premises,pass:premises&&referenceSecondSign===-1&&qcompare(mu,upper)<=0,
    actualSourceAuthenticated:false,absoluteMatrixNormSubstitutedForLogarithmicNorm:false};
}

function copyOuterFactorProgram(G,program,delta,eta){
  const map=new Map([[program.roots.outerLogDistance,delta],[program.roots.eta,eta]]),ops=new Set(),pairs=[];
  const copy=id=>{
    if(map.has(id))return map.get(id);const {op,args}=program.nodes[id];ops.add(op);let value;
    if(op==='rational')value=G.q(...args);
    else if(op==='source_parameter')value=G.parameter(args[0]);
    else if(op==='coordinate')value=G.fresh('outer_factor_import_'+args[0]);
    else if(['integer_power','source_step_derivative'].includes(op))value=G.node(op,[copy(args[0]),args[1]]);
    else if(['add','multiply','inverse','sqrt_positive','exp','log_positive','definite_integral','smooth_piecewise'].includes(op))value=G.node(op,args.map(copy));
    else fail('UNSUPPORTED','The actual outer flat factor contains an unsupported operation: '+op);
    map.set(id,value);pairs.push([id,value]);return value;
  };
  const roots=Object.fromEntries(['Xb','EpowB','rho','outerTheta','outerZ','outerScaledZ'].map(name=>[name,copy(program.roots[name])])),
    gamma=program.improper.find(x=>x.name==='actual_Euler_Gamma');
  if(!gamma)fail('INVALID_SOURCE_CONSTRUCTION','The full original Euler Gamma integral was omitted.');
  roots.gamma=copy(gamma.root);
  return {roots,copy,pairs,operations:[...ops].sort(),originalSchema:program.schema,
    originalParameterExpressionSHA256:program.parameterExpressionSHA256,sourceEquationList:[...program.sourceEquations],
    sourceKernelCount:program.outerKernels.length,sourceKernels:program.outerKernels.map(x=>({...x,root:copy(x.root)}))};
}
